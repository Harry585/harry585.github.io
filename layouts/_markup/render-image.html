{{- /* Image render hook: runs for every Markdown image, ![alt](src "title"). Images that live in the post's page bundle are resized to WebP with a srcset (which also strips EXIF data such as GPS location). Anything else (remote URLs, files in static/, GIFs/SVGs) is passed through unchanged. A "title" becomes a caption. */ -}} {{- $src := .Destination -}} {{- $img := .Page.Resources.Get $src -}} {{- $processable := false -}} {{- with $img -}} {{- $processable = in (slice "jpeg" "png" "webp" "tiff" "bmp") .MediaType.SubType -}} {{- end -}} {{- if .Title }}{{ end -}}
